
import React, { useState } from 'react';

export default function App() {
    const [candidates, setCandidates] = useState([
        { id: 1, name: "React", votes: 12 },
        { id: 2, name: "Vue.js", votes: 8 },
        { id: 3, name: "Angular", votes: 5 },
        { id: 4, name: "Svelte", votes: 7 }
    ]);

    const [voted, setVoted] = useState(false);
    const [name, setName] = useState("");

    const total = candidates.reduce((sum, c) => sum + c.votes, 0);

    function vote(id) {
        if (voted) return;

        setCandidates(candidates.map(c =>
            c.id === id ? { ...c, votes: c.votes + 1 } : c
        ));
        setVoted(true);
    }

    function addCandidate(e) {
        e.preventDefault();
        if (!name.trim()) return;

        setCandidates([...candidates, {
            id: Date.now(), name: name.trim(), votes: 0
        }]);
        setName("");
    }

    return (
        <div style={{ textAlign: "center", padding: 20 }}>
            <h1>Voting App</h1>

            {candidates.map(c => (
                <div key={c.id}>
                    <h3>{c.name} - {c.votes} votes</h3>
                    <button onClick={() => vote(c.id)} disabled={voted}>
                        Vote
                    </button>
                </div>
            ))}

            <h3>Total Votes: {total}</h3>

            <form onSubmit={addCandidate}>
                <input
                    value={name}
                    onChange={e => setName(e.target.value)}
                    placeholder="Candidate Name"
                />
                <button>Add</button>
            </form>

            <button onClick={() => setVoted(false)}>
                Vote Again
            </button>
        </div>
    );
}
