const input = document.querySelector("textarea");
const sendBtn = document.querySelector("#sendBTN");
const chatbox = document.querySelector(".chatbox");

// Create chat message
function createMessage(message, type) {
    const li = document.createElement("li");
    const p = document.createElement("p");

    li.className = "chat " + type;
    p.textContent = message;

    li.appendChild(p);
    return li;
}

// Send message to backend
async function generateResponse(message, chatMessage) {
    try {
        const response = await fetch("/api/chat", {
            method: "POST",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({ message })
        });

        if (!response.ok) {
            throw new Error("API error");
        }

        const data = await response.json();
        chatMessage.querySelector("p").textContent = data.reply;

    } catch (error) {
        chatMessage.querySelector("p").textContent =
            "Oops! Something went wrong.";

        chatMessage.querySelector("p").classList.add("error");
    }

    chatbox.scrollTop = chatbox.scrollHeight;
}

// Handle user message
function handleChat() {
    const message = input.value.trim();

    if (!message) return;

    chatbox.appendChild(createMessage(message, "chat-outgoing"));
    input.value = "";

    const botMessage = createMessage("Thinking...", "chat-incoming");
    chatbox.appendChild(botMessage);

    chatbox.scrollTop = chatbox.scrollHeight;

    generateResponse(message, botMessage);
}

// Send button
sendBtn.addEventListener("click", handleChat);

// Close chatbot
function cancel() {
    document.querySelector(".chatBot").style.display = "none";
}
