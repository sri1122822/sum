<script>
$("#password").keyup(function() {
    let p = $(this).val();
    let score = 0;

    if (p.length >= 6) score += 20;
    if (p.length >= 8) score += 20;
    if (/[a-z]/.test(p)) score += 15;
    if (/[A-Z]/.test(p)) score += 15;
    if (/[0-9]/.test(p)) score += 15;
    if (/\W/.test(p)) score += 15;

    $(".progress-bar")
        .css("width", score + "%")
        .css("background", score < 40 ? "red" : score < 70 ? "orange" : "green");
});
</script>
